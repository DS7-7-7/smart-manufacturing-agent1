[11.5 실습 1] MES/ERP 도구 단독 테스트
------------------------------------------
# LLM 없이 도구 5개가 잘 작동하는지 확인
py mes_tools.py

# 예상 출력:
#   [1] MES 생산 오더 조회 → 3개 오더 표시
#   [2] MES 설비 상태 조회 (3라인) → 온도, 진동 등
#   [3] ERP 재고 조회 (P-2201) → 재고 12개, 부족
#   [4] ERP 발주서 생성 → PO 번호 부여
#   [5] Slack 알림 → 발송 시뮬레이션


[11.6 실습 2] Function Calling 에이전트
------------------------------------------
# 대화형 실행
py agent_toolcall.py

# 1회성 요청
py agent_toolcall.py --once "3라인 지금 뭐 만들고 있어?"
python3 agent_toolcall.py --once "3라인 지금 뭐 만들고 있어?"

# 복합 요청 (도구 여러 개 조합)
py agent_toolcall.py --once "3라인 상태 확인하고 P-2201 재고도 알려줘"
python3 agent_toolcall.py --once "3라인 상태 확인하고 P-2201 재고도 알려줘"
# 최대 반복 조정
py agent_toolcall.py --max-steps 12


[11.7 실습 3] 자율 시나리오 데모
------------------------------------------
# 3가지 시나리오 모두 실행
py demo_scenario.py

# 특정 시나리오만 실행
py demo_scenario.py --scenario A     # 단일 조회
py demo_scenario.py --scenario B     # 조건부 알림
py demo_scenario.py --scenario C     # 자동 발주
